% Preamble for SaeedSalehi_CV.tex. Package order is deliberate: colour is
% available before the first \definecolor, and titlesec is patched only after
% it has redefined \section.

\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usepackage{microtype}

% amsmath is required for output fidelity, not for display math: it changes
% inline-math spacing, and without it the lines containing $\sim$ and $10,000$
% typeset a fraction of a point differently.
\usepackage{amsmath}

\usepackage{graphicx}
\usepackage{xcolor}
\usepackage[a4paper,margin=2cm, bottom=2cm]{geometry}
\definecolor{AccentColour}{HTML}{2C4A8C}
\definecolor{MarkerColour}{HTML}{2C4A8C}

\usepackage[colorlinks=true,linkcolor=blue,urlcolor=blue,citecolor=blue]{hyperref}
% PDF metadata: without these the Title/Author/Subject/Keywords fields are
% empty, so the file shows only as "SaeedSalehi_CV.pdf" in a browser tab,
% reference manager or document library.
\hypersetup{
  pdftitle={Saeed Salehi, Curriculum Vitae},
  pdfauthor={Saeed Salehi},
  pdfsubject={Curriculum vitae},
  pdfkeywords={computational fluid dynamics, machine learning, flow control,
    turbulence, uncertainty quantification, OpenFOAM, hydropower},
  pdflang={en},
}

\usepackage{enumitem}
\setlist[description]{leftmargin=0cm,labelindent=0cm}
\usepackage[fixed]{fontawesome5}

\setcounter{secnumdepth}{0} % No section numbering in CV
\frenchspacing
% Slack so an unbreakable token (a \texttt package name, a long URL) cannot
% push a line past the margin; at 12pt these would otherwise overflow.
\emergencystretch=3em

% \makefield{icon}{text} sets one contact line in Sections/Contact.tex.
\newcommand{\makefield}[2]{\makebox[1em]{\color{MarkerColour!80!black}#1} #2\hspace{1.5em}}
% \swref{bibkey} tags each item in Sections/OpenSource.tex with its bibliography
% key and prints nothing here. A document that reuses these section files with
% its own preamble may redefine it to print the corresponding citation.
\newcommand{\swref}[1]{}

\usepackage{titlesec}
\titleformat{\section}{
\vspace{1em}\scshape\raggedright\Large\color{AccentColour}
}{\thesection}{1em}{}[{\color{AccentColour}\titlerule[2pt]} \vspace{-2pt}]

\titleformat{\subsection}{
	\vspace{1em}\raggedright\bfseries\large
}{\thesubsection}{1em}{}

% With secnumdepth=0 every \section takes hyperref's unnumbered (\@ssect) path,
% which sets no anchor. Every bookmark therefore inherited the last anchor that
% did exist, so all 19 of them pointed at Doc-Start or section*.1.
% \phantomsection before each heading gives it its own destination. Loaded after
% titlesec so the patch applies to titlesec's redefinition.
\usepackage{etoolbox}
\pretocmd{\section}{\phantomsection}{}{\PackageError{CV}{patching \string\section\space failed}{}}
\pretocmd{\subsection}{\phantomsection}{}{\PackageError{CV}{patching \string\subsection\space failed}{}}
% Never strand a section heading at the bottom of a page: if less than about
% five lines remain (heading, rule, and the first lines of content), start the
% section on the next page.
\usepackage{needspace}
\pretocmd{\section}{\needspace{5\baselineskip}}{}{\PackageError{CV}{patching \string\section\space for needspace failed}{}}
\usepackage{bookmark}

\usepackage[maxbibnames=99, maxcitenames=2, defernumbers=true, doi=true, url=false, sorting=none, isbn=false]{biblatex}
\renewbibmacro{in:}{}
\addbibresource{Publications.bib}
% Titles link to the DOI when there is one, else to the URL, ISBN or ISSN.
\newbibmacro{string+doiurlisbn}[1]{%
    \iffieldundef{doi}{%
        \iffieldundef{url}{%
            \iffieldundef{isbn}{%
                \iffieldundef{issn}{%
                    #1%
                }{%
                    \href{http://books.google.com/books?vid=ISSN\thefield{issn}}{#1}%
                }%
            }{%
                \href{http://books.google.com/books?vid=ISBN\thefield{isbn}}{#1}%
            }%
        }{%
            \href{\thefield{url}}{#1}%
        }%
    }{%
        \href{https://doi.org/\thefield{doi}}{#1}%
    }%
}
\DeclareFieldFormat{title}{\usebibmacro{string+doiurlisbn}{\mkbibemph{#1}}}
\DeclareFieldFormat[article,incollection,inproceedings]{title}%
{\usebibmacro{string+doiurlisbn}{\mkbibquote{#1}}}
\AtEveryBibitem{\clearfield{month}}
\AtEveryCitekey{\clearfield{month}}
% addendum carries two different things in Publications.bib: the CRediT
% statement on journal papers and the book chapter, and the conference name and
% location on the @unpublished presentations. Clearing it unconditionally hid
% the CRediT as intended but also stripped every presentation of its conference
% and venue, leaving only a title and a year. Keep it for @unpublished.
\AtEveryBibitem{\ifentrytype{unpublished}{}{\clearfield{addendum}}}
\AtEveryCitekey{\ifentrytype{unpublished}{}{\clearfield{addendum}}}

% Bold the author's own name, which Publications.bib annotates as "highlight".
\renewcommand*{\mkbibnamegiven}[1]{%
  \ifitemannotation{highlight}
    {\textbf{#1}}
    {#1}}

\renewcommand*{\mkbibnamefamily}[1]{%
  \ifitemannotation{highlight}
    {\textbf{#1}}
    {#1}}
