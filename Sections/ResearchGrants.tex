% Replaces the previous run-in description list, whose italic labels
% ("Funding agency:", "Time period:", ...) sat inline so no two values ever
% lined up. Here the labels occupy a fixed column, so funders, periods, amounts
% and roles align vertically down the whole section and can be compared at a
% glance. The title carries the weight; the role value is emphasised only where
% it is PI, which is the distinction a reader is looking for.
%
% Everything is set at the CV's body size except the titles, which are small
% bold: labels are marked by colour, not by being shrunk. The \strut gives the
% label and value boxes a common first baseline, and the \nobreak keeps a
% grant's metadata block from being split across a page break.
%
% Defined here with \providecommand rather than in Includes/Settings.tex so the
% same file also works when it is \input by the Zenith proposal appendix, which
% supplies its own preamble and does not read the CV's Settings.tex.
\providecommand{\grantfield}[2]{%
  \nobreak\noindent\hspace*{1.3em}%
  \parbox[t]{7.25em}{\raggedright\strut{\color{AccentColour!85}#1}}%
  \parbox[t]{\dimexpr\linewidth-8.5em\relax}{\raggedright\strut #2}%
  \par\nobreak\vspace{0pt}}
\providecommand{\grantentry}[6]{%
  \par\addvspace{1.6em}%
  \noindent{\small\bfseries\color{AccentColour}#1}\par
  \nobreak\vspace{0.15em}%
  \grantfield{Funding agency}{#2}%
  \grantfield{Time period}{#3}%
  \grantfield{Amount}{#4}%
  \grantfield{Role}{#5}%
  \vspace{0.22em}%
  \begingroup\leftskip=1.3em\relax\noindent #6\par\endgroup
}
